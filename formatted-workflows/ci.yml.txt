name: CI

on:
  pull_request:
  push:
    branches:
      - main

permissions:
  contents: read

jobs:
  validate:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Setup Bun
        uses: oven-sh/setup-bun@v2
        with:
          bun-version: '1.3.13'

      - name: Install dependencies
        run: bun install --frozen-lockfile

      - name: Run build
        run: bun run build

      - name: Run typecheck
        run: bun run typecheck

      - name: Run lint
        run: bun run lint

      - name: Run format check
        run: bun run format:check

      - name: Run tests
        run: bun run test

      - name: Run Knip
        run: bun run knip

      - name: Generate docs
        run: bun run docs

      - name: Verify generated docs are committed
        run: git diff --exit-code

      - name: Run Doctor validation
        run: bunx @ankhorage/doctor@0.7.0 validate .

      - name: Check changesets
        if: github.event_name == 'pull_request'
        run: bash ./scripts/check-changeset-status.sh
